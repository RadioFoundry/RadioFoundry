# Facts for the stack check (tools/stack_check.py, docs/dev/stack-check.md) that the call graph and the sources
# don't show. One file for every board: a target a build doesn't have is ignored.
#
#   task FUNCTION STACK          a task the xTaskCreate scan can't find (STACK: words, a number or a macro)
#   ptr NAME TARGET...           calls through the pointer field, variable or parameter NAME
#   call FUNCTION TARGET...      every indirect call FUNCTION makes
#   file PATH TARGET...          indirect calls in PATH that the sources don't resolve
#   lib FUNCTION BYTES           the stack of a function without call graph (assembly, the C library)
#   extern NAME BYTES            code outside the build, a target written @NAME
#   recursion FUNCTION...        recursion known to be bounded (not reported)
#
# A TARGET is a function name, or path/file.c:name for a static one.

# FreeRTOS creates its idle task through a variable (tasks.c)
task prvIdleTask configMINIMAL_STACK_SIZE

# The stock AMBE+2 codec runs on the calling task's stack. STM32: the largest stock codec's peak, 4046 bytes
# measured (docs/spec/codec-md380.md), plus the veneer's 20 bytes of registers, up to 8 stack arguments
# and 4 for alignment (platform/stm32f4/codec_call.c, assembly)
lib stm32_codec_call 4104
# HR_C7000: the same codec library on C-SKY, called by address (platform/hrc7000/vocoder.c). Not measured on this core:
# the STM32 figure, rounded up
extern hrc_codec 4096
file platform/hrc7000/vocoder.c @hrc_codec

# The stub entry (core/stubentry) jumps into the flash stub with interrupts off; ODR never resumes, so what the stub does
# with the stack doesn't matter
extern flash_stub 0
call enter_stub @flash_stub

# A board whose EEPROM glue wraps I2C bus A's locked calls in an i2c_bus_t of its own (boards/gd77/eeprom_gd77.c): the
# bus calls look like a cycle through i2c_bus_t. Bus A itself is the software master: there is no recursion
recursion i2c_a_write i2c_a_write_read ee_bus_write ee_bus_write_read

# The C library, where its worst path needs arguments ODR never passes. Each cuts that function out of the disassembly
# walk: check the reason still holds when adding a call.
# vPortYield (C-SKY port, assembly): saves the 17-word frame the port overhead already counts
lib vPortYield 0
# newlib's strstr takes its table-driven path (1 KB of stack on C-SKY) only for needles of 32 characters or more;
# every strstr in ODR searches for a short literal
lib two_way_long_needle 0
# Floating point conversions (%e %f %g) only: ODR formats with %d and %s (snprintf in core/dmr). Without them the
# C-SKY newlib's snprintf needs about 0.6 KB, with them 2 KB
lib _dtoa_r 0
