-- The RadioFoundry enterprise MIB (docs/spec/snmp.md). The enterprise number 32473 is the one IANA reserves for
-- documentation (RFC 5612): a placeholder until the project has its own Private Enterprise Number. When it is
-- assigned, change it here, in core/snmp/snmpMib.h (RFOUNDRY_SNMP_PEN) and in tools/rfoundry_snmp.py (PEN).

RFOUNDRY-MIB DEFINITIONS ::= BEGIN

IMPORTS
    MODULE-IDENTITY, OBJECT-TYPE, NOTIFICATION-TYPE, enterprises, Integer32, Unsigned32, Counter32, TimeTicks
        FROM SNMPv2-SMI
    DisplayString, TruthValue
        FROM SNMPv2-TC
    MODULE-COMPLIANCE, OBJECT-GROUP, NOTIFICATION-GROUP
        FROM SNMPv2-CONF;

rfoundryMIB MODULE-IDENTITY
    LAST-UPDATED "202610110000Z"
    ORGANIZATION "The RadioFoundry Authors"
    CONTACT-INFO "https://github.com/RadioFoundry"
    DESCRIPTION
        "Objects and notifications of a radio running RadioFoundry firmware: identity, power, the channel and
        the DMR call state, the DMR data service, the USB network link and the agent's own settings."
    REVISION     "202610110000Z"
    DESCRIPTION  "First version."
    ::= { rfoundry 2 }

-- The agent's settings (saved in the radio)

rfoundryAgentRfManager OBJECT-TYPE
    SYNTAX      Unsigned32 (0..16777215)
    MAX-ACCESS  read-write
    STATUS      current
    DESCRIPTION
        "The DMR ID notifications are sent to over the air (to UDP 162 at its 12.x.y.z address), 0 = none

rfoundryAgentRfTrapInterval OBJECT-TYPE
    SYNTAX      Unsigned32 (10..3600)
    UNITS       "seconds"
    MAX-ACCESS  read-write
    STATUS      current
    DESCRIPTION
        "At most one notification over the air per this many seconds; the others are dropped. Emergency
        notifications are exempt. Default 60."
    ::= { rfoundryAgent 2 }

rfoundryAgentRfAnswers OBJECT-TYPE
    SYNTAX      TruthValue
    MAX-ACCESS  read-write
    STATUS      current

    SYNTAX      TruthValue

rfoundryAgentUsbTraps OBJECT-TYPE

rfoundryAgentRfCallTraps OBJECT-TYPE
    SYNTAX      TruthValue
    MAX-ACCESS  read-write

rfoundryAgentTrapsSent OBJECT-TYPE
    SYNTAX      Counter32

rfoundryAgentTrapsDropped OBJECT-TYPE
    SYNTAX      Counter32
    MAX-ACCESS  read-only
    STATUS      current

        rfoundryDmrId, rfoundryCallsign, rfoundryBoard, rfoundryFirmware,

        rfoundryTxFrequency, rfoundryTxPowerLevel, rfoundryTxPowerName, rfoundryColorCode, rfoundryTimeslot, rfoundryTalkTarget, rfoundryTalkTargetType,
        rfoundryRole,
        rfoundryCallState, rfoundryCallSource, rfoundryCallDestination, rfoundryLastSource, rfoundryLastDestination, rfoundryLastGroup, rfoundryLastAge,

        rfoundryUsbNetwork, rfoundryNetToAir, rfoundryNetPacketsUp
